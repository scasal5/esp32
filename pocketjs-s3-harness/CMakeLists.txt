cmake_minimum_required(VERSION 3.22)
set(IDF_TARGET esp32s3 CACHE STRING "")
set(HARNESS_MODE native CACHE STRING "probe, native, headless, pocket")
set(ENV{WS183_BUILD_MODE} "${HARNESS_MODE}")
set(POCKETJS_ROOT "${CMAKE_CURRENT_LIST_DIR}/.deps/pocketjs" CACHE PATH "Pinned checkout")
set(EXTRA_COMPONENT_DIRS "${CMAKE_CURRENT_LIST_DIR}/../firmware/components/xpowerslib")
if(HARNESS_MODE STREQUAL "headless" OR HARNESS_MODE STREQUAL "pocket")
    foreach(component pocketjs_package pocketjs_guest)
        list(APPEND EXTRA_COMPONENT_DIRS "${POCKETJS_ROOT}/hosts/esp-idf/components/${component}")
    endforeach()
endif()
if(HARNESS_MODE STREQUAL "pocket")
    foreach(component pocketjs_ui_core pocketjs_ui_qjs pocketjs_render_rgb565)
        list(APPEND EXTRA_COMPONENT_DIRS "${POCKETJS_ROOT}/hosts/esp-idf/components/${component}")
    endforeach()
endif()
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
idf_build_set_property(MINIMAL_BUILD ON)
project(ws183_harness)
